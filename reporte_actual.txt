// Listar archivos guardados en orden cronológico (por nombre)
vector<string> listarArchivosGuardados() {
    vector<string> archivos;

    namespace fs = std::filesystem;

    for (const auto& entrada : fs::directory_iterator(".")) {
        string nombre = entrada.path().filename().string();
        if (nombre.size() == 19 && nombre[8] == '-' && nombre.substr(15) == ".txt" &&
            nombre.substr(0, 8).find_first_not_of("0123456789") == string::npos &&
            nombre.substr(9, 6).find_first_not_of("0123456789") == string::npos) {
            archivos.push_back(nombre);
        }
    }

    // Ordenar alfabéticamente = orden cronológico por el patrón de nombre
    std::sort(archivos.begin(), archivos.end());

    return archivos;
}

// Función auxiliar para validar lógicamente una matriz Sudoku
bool sudokuValido(int matriz[TAM][TAM]) {
    // Validar filas
    for (int fila = 0; fila < TAM; fila++) {
        bool vistos[10] = {false};
        for (int col = 0; col < TAM; col++) {
            int val = matriz[fila][col];
            if (val == 0)
                continue;
            if (vistos[val])
                return false;
            vistos[val] = true;
        }
    }

    // Validar columnas
    for (int col = 0; col < TAM; col++) {
        bool vistos[10] = {false};
        for (int fila = 0; fila < TAM; fila++) {
            int val = matriz[fila][col];
            if (val == 0)
                continue;
            if (vistos[val])
                return false;
            vistos[val] = true;
        }
    }

    // Validar subcuadros 3x3
    for (int baseFila = 0; baseFila < TAM; baseFila += 3) {
        for (int baseCol = 0; baseCol < TAM; baseCol += 3) {
            bool vistos[10] = {false};
            for (int f = 0; f < 3; f++) {
                for (int c = 0; c < 3; c++) {
                    int val = matriz[baseFila + f][baseCol + c];
                    if (val == 0)
                        continue;
                    if (vistos[val])
                        return false;
                    vistos[val] = true;
                }
            }
        }
    }

    // Si pasó todo, es válido
    return true;
}

// ----------------------------------------------------------------
// Carga los scores desde ARCHIVO_SCORES y los ordena en 'scores'
// ----------------------------------------------------------------
void cargarScores() {
    // 0) Asegurar existencia del archivo
    if (!std::filesystem::exists(ARCHIVO_SCORES)) {
        std::ofstream f(ARCHIVO_SCORES);
        f.close();
    }

    // 1) Inicializar con entradas “vacías” (errores y tiempos muy altos)
    for (int lvl = 0; lvl < 3; ++lvl) {
        for (int i = 0; i < 5; ++i) {
            scores[lvl][i] = ScoreEntry{
                "---",            // iniciales
                lvl,              // nivel
                INT_MAX,          // erroresFallidos
                INT_MAX,          // duracionSegundos
                endGame,          // erroresPermitidos
                "00000000-000000" // timestamp
            };
        }
    }

    // 2) Leer y parsear cada línea del archivo
    std::ifstream f(ARCHIVO_SCORES);
    if (!f.is_open())
        return;

    std::string linea;
    while (std::getline(f, linea)) {
        if (linea.empty())
            continue;
        // 2.1) Ignorar encabezados de nivel (comienzan con "+++")
        if (linea.rfind("+++", 0) == 0)
            continue;

        // 2.2) Leer los 6 campos separados por espacios
        std::istringstream iss(linea);
        std::string iniStr, timestamp;
        int lvl, erroresFall, durSeg, erroresPermit;
        if (!(iss >> iniStr >> lvl >> erroresFall >> durSeg >> erroresPermit >> timestamp)) {
            // línea mal formateada
            continue;
        }

        // 3) Mapear a ScoreEntry
        ScoreEntry e;
        // Iniciales (hasta 3 caracteres)
        for (int k = 0; k < 3; ++k)
            e.iniciales[k] = (k < (int)iniStr.size() ? toupper(iniStr[k]) : ' ');
        e.iniciales[3] = '\0';

        e.nivel = lvl;
        e.erroresFallidos = erroresFall;
        e.duracionSegundos = durSeg;
        e.erroresPermitidos = erroresPermit; // el nuevo campo

        // Timestamp
        std::strncpy(e.timestamp, timestamp.c_str(), sizeof(e.timestamp));
        e.timestamp[sizeof(e.timestamp) - 1] = '\0';

        // 4) Insertar en la matriz ordenada
        actualizarScores(e);
    }
}

// ----------------------------------------------------------------
// Inserta una nueva entrada en 'scores', manteniendo orden y top 5
// ----------------------------------------------------------------
void actualizarScores(const ScoreEntry& entrada) {
    int lvl = entrada.nivel;
    if (lvl < 0 || lvl > 2)
        return;
    for (int i = 0; i < 5; ++i) {
        // Encuentra la posición donde entra la nueva entrada
        if (entrada.erroresFallidos < scores[lvl][i].erroresFallidos ||
            (entrada.erroresFallidos == scores[lvl][i].erroresFallidos &&
             entrada.duracionSegundos < scores[lvl][i].duracionSegundos)) {
            // Desplaza todos los elementos desde el final hasta i
            for (int j = 4; j > i; --j) {
                scores[lvl][j] = scores[lvl][j - 1];
            }
            // Inserta la nueva entrada en la posición i
            scores[lvl][i] = entrada;
            break;
        }
    }
}

// ----------------------------------------------------------------
// Guarda la matriz 'scores' en ARCHIVO_SCORES
// ----------------------------------------------------------------
void guardarScores() {
    std::ofstream f(ARCHIVO_SCORES);
    if (!f.is_open()) {
        mostrarError("No se pudo abrir el archivo de scores.");
        return;
    }

    // Para cada nivel, de 0 a 2
    for (int lvl = 0; lvl < 3; ++lvl) {
        // Decoración de nivel
        const char* nivelStr = (lvl == 0   ? "+++ FACIL +++"
                                : lvl == 1 ? "+++ MEDIO +++"
                                           : "+++ DIFICIL +++");
        f << nivelStr << '\n';

        // Entradas del ranking
        for (int i = 0; i < 5; ++i) {
            const ScoreEntry& e = scores[lvl][i];
            // Línea: iniciales - nivel - erroresFallidos - duracionSegundos - erroresPermitidos - timestamp
            f << e.iniciales << ' '
              << e.nivel << ' '
              << e.erroresFallidos << ' '
              << e.duracionSegundos << ' '
              << e.erroresPermitidos << ' '
              << e.timestamp << '\n';
        }
    }

    f.close();
}

// ----------------------------------------------------------------
// Resetea los 5 mejores scores por nivel y sobrescribe el archivo
// ----------------------------------------------------------------
void resetScores() {
    for (int lvl = 0; lvl < 3; ++lvl) {
        for (int i = 0; i < 5; ++i) {

            scores[lvl][i] = ScoreEntry{
                "---",            // iniciales
                lvl,              // nivel
                INT_MAX,          // erroresFallidos
                INT_MAX,          // duracionSegundos
                endGame,          // erroresPermitidos   ← ahora lo incluimos
                "00000000-000000" // timestamp
            };
        }
    }
    guardarScores();
}

                MSGPAUSA();

        case IngresoStatus::BACKDOOR: {

            if (nivelBD != -1)
                subMenuBackDoor(nivelBD);
            limpiarMascara(MASCARA);
            continue;
        }
        case IngresoStatus::CANCELAR:
            limpiarMascara(MASCARA);
            continue;
        case IngresoStatus::ERROR_SALIR:
            limpiarMascara(MASCARA);
            return;

    resetColor();

    if (nivelCargado >= 0) {
        // Ya hay nivel cargado en formato indice 0..2 (posible reanudada)
        nivelIndex = nivelCargado;
        // ⚠ No tocamos esPartidaNueva aqui: lo seteara cargarPartidaDesdeArchivo()
    } else {
        // Pedimos 0..3 (0 = cancelar, 1..3 = opcion)

        // Cancelacion
        if (opcion == CODIGO_SALIR_MENU) {
            colorTexto("amarillo");
            std::cout << "Operacion cancelada. Volviendo al menu..." << std::endl;
            resetColor();
            MSGPAUSA();
            return;
        }

        // Convertimos 1..3 a indice 0..2
        nivelIndex = opcion - 1;

        // ✅ Partida nueva (seleccion por menu)
        esPartidaNueva = true;
    }

    // Seguridad
    if (nivelIndex < 0 || nivelIndex > 2) {
        mostrarError("Nivel invalido.");
        return;
    }

    // Persistimos nivel y marcamos partida (si corresponde)
    nivelCargado = nivelIndex;
    // partidaEnCurso = true; // descomentar si este es el momento de marcarla

    // Mensaje usuario
    std::string nombreDificultad;
    switch (nivelIndex) {
    case 0:
        nombreDificultad = "FACIL";
        break;
    case 1:
        nombreDificultad = "MEDIA";
        break;
    case 2:
        nombreDificultad = "DIFICIL";
        break;
    default:
        nombreDificultad = "DESCONOCIDA";
        break;
    }
    std::cout << "\nDificultad seleccionada: " << nombreDificultad << std::endl;

    // Arranca la partida
    jugarPartida(nivelIndex);
}

    switch (nivel) {
    case 0:
        nombreDificultad = "Fácil";
        break;
    case 1:
        nombreDificultad = "Media";
        break;
    case 2:
        nombreDificultad = "Difícil";
        break;
    default:
        nombreDificultad = "Desconocida";
        break;
    }

    colorTexto("cyan");
    cout << "📘 Nivel seleccionado: " << nombreDificultad << " (" << nivel << ")" << endl;
    resetColor();

    if (nuevaPartida) {
        intentosFallidos = 0;
    }

    if (!esReanudar) {
        generarSodokuResuelto();
        generarTableroAleatorio(nivel);
        fijarCeldasActuales();
        partidaEnCurso = true;
    }

    // ⏱️ tiempos
    time_t tiempoInicio = time(nullptr);
    tm* inicio = localtime(&tiempoInicio);
    tsSesionInicio = tiempoInicio; // acumulo sesion

    int numerosAgotados = 0;
    bool seguir = true;
    while (seguir) {
        limpiarPantalla();
        autor();
        DibujarPantalla();
        bool completo = true;

        // GAME OVER
        if (endGame > 0 && intentosFallidos >= endGame) {
            cout << "❌ Intentos fallidos: " << intentosFallidos
                 << "   |   🔢 Números agotados: " << numerosAgotados << endl
                 << endl;

            colorTexto("rojo");
            cout << "JUEGO TERMINADO" << endl;
            resetColor();

            // resumen de sesion
            time_t tiempoFin = time(nullptr);
            tm* fin = localtime(&tiempoFin);
            int duracion = static_cast<int>(difftime(tiempoFin, tiempoInicio));
            int minutos = duracion / 60, segundos = duracion % 60;
            char horaInicioStr[9], horaFinStr[9];
            strftime(horaInicioStr, sizeof(horaInicioStr), "%H:%M:%S", inicio);
            strftime(horaFinStr, sizeof(horaFinStr), "%H:%M:%S", fin);

            colorTexto("cyan");
            cout << "\n⏱️ Hora de inicio  : " << horaInicioStr << endl;
            cout << "⏱️ Hora de fin     : " << horaFinStr << endl;
            cout << "⏱️ Duración total  : " << minutos << " min " << segundos << " seg." << endl;
            cout << "🔢 Números agotados  : " << numerosAgotados << endl;
            cout << "❌ Intentos fallidos : " << intentosFallidos << endl;
            resetColor();

            duracionAcumulada += static_cast<long>(difftime(time(nullptr), tsSesionInicio));
            partidaEnCurso = false;
            MSGPAUSA();
            break;
        }

        cout << "❌ Intentos fallidos: " << intentosFallidos
             << "   |   🔢 Números agotados: " << numerosAgotados << endl
             << endl;
        cout << "Cero (0) en cualquier momento para salir." << endl;

        int fila, columna, valor;
        IngresoStatus status = procesarIngresoUsuario(fila, columna, valor);
        switch (status) {
        case IngresoStatus::OK: {
            // 🔒 nunca escribir en celdas fijas
            if (CELDAFIJA[fila - 1][columna - 1]) {
                mostrarError("Celda fija: no se puede modificar.");
                limpiarMascara(MASCARA);
                continue;
            }

            // escribir valor
            MATRIZ[fila - 1][columna - 1] = valor;

            if (cantidadRestante(valor) == 0) {
                numerosAgotados++;
                colorTexto("rojo");
                cout << "💥 ¡Atención! El número " << valor << " ha sido agotado." << endl;
                resetColor();
                MSGPAUSA();
            }
            limpiarMascara(MASCARA);
            break;
        }
        case IngresoStatus::BACKDOOR: {
            int nivelBD = validarNivelDeAcceso();
            if (nivelBD != -1)
                subMenuBackDoor(nivelBD);
            limpiarMascara(MASCARA);
            continue;
        }
        case IngresoStatus::CANCELAR:
            limpiarMascara(MASCARA);
            continue;
        case IngresoStatus::ERROR_SALIR:
            duracionAcumulada += static_cast<long>(difftime(time(nullptr), tsSesionInicio));
            seguir = false;
            return;
        }

        // Chequeo de victoria
        for (int i = 0; i < TAM && completo; ++i) {
            for (int j = 0; j < TAM; ++j) {
                if (MATRIZ[i][j] != SOLUCION[i][j]) {
                    completo = false;
                    break;
                }
            }
        }
        if (completo) {
            colorTexto("verde");
            cout << "🎉 ¡Felicitaciones! Completaste el Sudoku." << endl;
            resetColor();

            time_t tiempoFin = time(nullptr);
            tm* fin = localtime(&tiempoFin);
            int duracion = static_cast<int>(difftime(tiempoFin, tiempoInicio));
            int minutos = duracion / 60, segundos = duracion % 60;
            char horaInicioStr[9], horaFinStr[9];
            strftime(horaInicioStr, sizeof(horaInicioStr), "%H:%M:%S", inicio);
            strftime(horaFinStr, sizeof(horaFinStr), "%H:%M:%S", fin);

            colorTexto("cyan");
            cout << "\n⏱️ Hora de inicio  : " << horaInicioStr << endl;
            cout << "⏱️ Hora de fin     : " << horaFinStr << endl;
            cout << "⏱️ Duración total  : " << minutos << " min " << segundos << " seg." << endl;
            cout << "🔢 Números agotados  : " << numerosAgotados << endl;
            cout << "❌ Intentos fallidos : " << intentosFallidos << endl;
            resetColor();

            duracionAcumulada += static_cast<long>(difftime(time(nullptr), tsSesionInicio));
            partidaEnCurso = false;

            // (scores retro)  —— sin cambios
            char inic[4];
            std::string iniciales;
            do {
                cout << "Ingresá tus 3 iniciales (A–Z, 0–9): ";
                if (!(std::cin >> iniciales))
                    return;
                if (iniciales.size() != 3)
                    std::cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n');
            } while (iniciales.size() != 3);
            for (int k = 0; k < 3; ++k)
                inic[k] = std::toupper(static_cast<unsigned char>(iniciales[k]));
            inic[3] = '\0';

            ScoreEntry nueva;
            strncpy(nueva.iniciales, inic, 4);
            nueva.nivel = nivel;
            nueva.erroresFallidos = intentosFallidos;
            nueva.duracionSegundos = static_cast<int>(difftime(time(nullptr), tiempoInicio));
            nueva.erroresPermitidos = endGame;
            {
                time_t ahora = time(nullptr);
                tm* t = localtime(&ahora);
                strftime(nueva.timestamp, sizeof(nueva.timestamp), "%Y%m%d-%H%M%S", t);
            }
            cargarScores();
            actualizarScores(nueva);
            guardarScores();
            mostrarScores(nivel);

            MSGPAUSA();
            break;
        }
    }
}

bool validaHorizontal(int matriz[TAM][TAM], int fila, int valor) {
    for (int j = 0; j < TAM; j++) {
        if (matriz[fila][j] == valor)
            return false;
    }
    return true;
}
//---------------------------------------------------------------------------------------
bool validaVertical(int matriz[TAM][TAM], int columna, int valor) {
    for (int i = 0; i < TAM; i++) {
        if (matriz[i][columna] == valor)
            return false;
    }
    return true;
}
//---------------------------------------------------------------------------------------
bool validaRecuadro(int matriz[TAM][TAM], int fila, int columna, int valor) {
    int filaInicio = (fila / 3) * 3;
    int columnaInicio = (columna / 3) * 3;
    for (int i = filaInicio; i < filaInicio + 3; i++) {
        for (int j = columnaInicio; j < columnaInicio + 3; j++) {
            if (matriz[i][j] == valor)
                return false;
        }
    }
    return true;
}
//---------------------------------------------------------------------------------------
void fijarCeldasActuales() {
    for (int i = 0; i < TAM; i++)
        for (int j = 0; j < TAM; j++)
            if (MATRIZ[i][j] != 0)
                CELDAFIJA[i][j] = true;

    mostrarError("Celdas actuales fijadas correctamente.");
}
//---------------------------------------------------------------------------------------
bool hayCeldasFijas() {
    for (int i = 0; i < TAM; i++)
        for (int j = 0; j < TAM; j++)
            if (CELDAFIJA[i][j])
                return true;
    return false;
}
//---------------------------------------------------------------------------------------
bool numeroAgotadoGlobal(int numero) {
    int contador = 0;
    for (int fila = 0; fila < TAM; fila++) {
        for (int columna = 0; columna < TAM; columna++) {
            if (MATRIZ[fila][columna] == numero) {
                contador++;
                if (contador >= TAM)
                    return true;
            }
        }
    }
    return false;
}
//---------------------------------------------------------------------------------------
bool numeroPresenteEnCuadro(int fila, int columna, int numero) {
    int filaInicio = (fila / 3) * 3;
    int columnaInicio = (columna / 3) * 3;

    for (int i = filaInicio; i < filaInicio + 3; i++) {
        for (int j = columnaInicio; j < columnaInicio + 3; j++) {
            if (MATRIZ[i][j] == numero)
                return true;
        }
    }

    return false;
}
//---------------------------------------------------------------------------------------
int cantidadRestante(int numero) {
    int contador = 0;

    for (int fila = 0; fila < TAM; fila++) {
        for (int columna = 0; columna < TAM; columna++) {
            if (MATRIZ[fila][columna] == numero)
                contador++;
        }
    }

    return TAM - contador;
}

//---------------------------------------------------------------------------------------
IngresoStatus procesarIngresoUsuario(int& fila, int& columna, int& valor) {
    // Fila
    cout << "Ingrese la fila (1 a " << TAM << "): ";
    cin >> fila;
    if (cin.fail() || fila < 0 || fila > TAM) {
        cin.clear();
        cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n');
        mostrarError("Fila inválida.");
        limpiarMascara(MASCARA);
        return IngresoStatus::CANCELAR; // no cuenta intento
    }
    if (fila == 0)
        return IngresoStatus::ERROR_SALIR;

    marcarFila(fila - 1);
    limpiarPantalla();
    autor();
    DibujarPantalla();

    // Columna
    cout << "Ingrese la columna (1 a " << TAM << "): ";
    cin >> columna;
    if (cin.fail() || columna < 0 || columna > TAM) {
        cin.clear();
        cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n');
        mostrarError("Columna inválida.");
        limpiarMascara(MASCARA);
        return IngresoStatus::CANCELAR; // no cuenta intento
    }
    if (columna == 0)
        return IngresoStatus::ERROR_SALIR;

    marcarColumna(columna - 1);
    marcarInterseccion(fila - 1, columna - 1);
    limpiarPantalla();
    autor();
    DibujarPantalla();

    // Si la celda es fija, no permitir (defensa en profundidad)
    if (CELDAFIJA[fila - 1][columna - 1]) {
        mostrarError("Celda fija: no se puede modificar.");
        limpiarMascara(MASCARA);
        return IngresoStatus::CANCELAR; // no cuenta intento
    }

    // Si ya tiene valor, menu contextual (sin contar intentos)
    if (MATRIZ[fila - 1][columna - 1] != 0) {
        colorTexto("amarillo");
        cout << "⚠️  Atención: esa celda ya contiene un valor: "
             << MATRIZ[fila - 1][columna - 1] << endl;
        resetColor();
        cout << "[1] Modificar valor\n[2] Borrar valor (volver a '-')\n[0] Cancelar\nSeleccione una opción: ";
        int decision;
        cin >> decision;
        if (cin.fail() || decision < 0 || decision > 2) {
            cin.clear();
            cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n');
            mostrarError("Opción inválida.");
            limpiarMascara(MASCARA);
            return IngresoStatus::CANCELAR;
        }
        if (decision == 0) {
            limpiarMascara(MASCARA);
            return IngresoStatus::CANCELAR;
        }
        if (decision == 2) {
            MATRIZ[fila - 1][columna - 1] = 0;
            limpiarMascara(MASCARA);
            return IngresoStatus::CANCELAR;
        }
        // decision==1 -> seguimos
    }

    // Valor
    cout << "Ingrese valor (1-" << TAM
         << ", " << CODIGO_SALIR << " = cancelar, "
         << CODIGO_BACKDOOR << " = menú oculto): ";
    cin >> valor;
    if (cin.fail()) {
        cin.clear();
        cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n');
        mostrarError("Valor inválido.");
        limpiarMascara(MASCARA);
        return IngresoStatus::CANCELAR; // no cuenta intento
    }

    if (valor == CODIGO_BACKDOOR)
        return IngresoStatus::BACKDOOR;
    if (valor == CODIGO_SALIR)
        return IngresoStatus::CANCELAR;
    if (valor == CODIGO_SALIR_MENU)
        return IngresoStatus::ERROR_SALIR;

    if (valor < 1 || valor > TAM) {
        mostrarError("Valor fuera de rango.");
        limpiarMascara(MASCARA);
        return IngresoStatus::CANCELAR; // no cuenta intento
    }

    // ✅ Debe coincidir con la solucion (si disponible)
    {
        int esperado = SOLUCION[fila - 1][columna - 1];
        if (esperado != 0 && valor != esperado) {
            mostrarError("Valor incorrecto segun la solucion.");
            ++intentosFallidos; // cuenta intento (regla)
            limpiarMascara(MASCARA);
            return IngresoStatus::CANCELAR;
        }
    }

    // Reglas de Sudoku (cuentan intento)
    if (numeroAgotadoGlobal(valor)) {
        mostrarError("🚫 Número ya fue utilizado 9 veces.");
        ++intentosFallidos;
        limpiarMascara(MASCARA);
        return IngresoStatus::CANCELAR;
    }
    if (numeroPresenteEnCuadro(fila - 1, columna - 1, valor)) {
        mostrarError("⚠️ Ese número ya está presente en este subcuadro 3x3.");
        ++intentosFallidos;
        limpiarMascara(MASCARA);
        return IngresoStatus::CANCELAR;
    }

    bool hayError = false;
    if (!validaHorizontal(MATRIZ, fila - 1, valor)) {
        mostrarError("Error: El valor ya existe en esa fila.");
        hayError = true;
    }
    if (!validaVertical(MATRIZ, columna - 1, valor)) {
        mostrarError("Error: El valor ya existe en esa columna.");
        hayError = true;
    }
    if (!validaRecuadro(MATRIZ, fila - 1, columna - 1, valor)) {

        limpiarMascara(MASCARA);
        return IngresoStatus::CANCELAR;
    }

    return IngresoStatus::OK;
}

void vaciarCeldas(int nivel) {
    int celdasAVaciar;
    switch (nivel) {
    case 0:
        celdasAVaciar = VACIOS_FACIL;
        break;
    case 1:
        celdasAVaciar = VACIOS_MEDIA;
        break;
    case 2:
        celdasAVaciar = VACIOS_DIFICIL;
        break;
    default:
        celdasAVaciar = VACIOS_MEDIA;
        break;
    }

    // 1) Construir lista de todas las posiciones
    std::vector<std::pair<int, int>> posiciones;
    posiciones.reserve(TAM * TAM);
    for (int i = 0; i < TAM; ++i)
        for (int j = 0; j < TAM; ++j)
            posiciones.emplace_back(i, j);

    // 2) Mezclar con mt19937
    std::shuffle(posiciones.begin(), posiciones.end(), rng);

        int i = posiciones[k].first;
        int j = posiciones[k].second;
        MATRIZ[i][j] = 0;

              << " celdas para la dificultad seleccionada." << std::endl;
    resetColor();

bool esSeguro(int matriz[TAM][TAM], int fila, int columna, int numero) {
    // Verificar fila
    for (int col = 0; col < TAM; col++)
        if (matriz[fila][col] == numero)
            return false;

    // Verificar columna
    for (int fil = 0; fil < TAM; fil++)
        if (matriz[fil][columna] == numero)
            return false;

    // Verificar subcuadro 3x3
    int inicioFila = fila - fila % 3;
    int inicioCol = columna - columna % 3;
    for (int i = 0; i < 3; i++)
        for (int j = 0; j < 3; j++)
            if (matriz[inicioFila + i][inicioCol + j] == numero)
                return false;

    return true;
}

bool resolverSudoku(int matriz[TAM][TAM]) {
    for (int fila = 0; fila < TAM; fila++) {
        for (int columna = 0; columna < TAM; columna++) {
            if (matriz[fila][columna] == 0) {
                std::vector<int> numeros = {1, 2, 3, 4, 5, 6, 7, 8, 9};
                // Nuevo shuffle con mt19937
                std::shuffle(numeros.begin(), numeros.end(), rng);
                for (int num : numeros) {
                    if (esSeguro(matriz, fila, columna, num)) {
                        matriz[fila][columna] = num;
                        if (resolverSudoku(matriz))
                            return true;
                        matriz[fila][columna] = 0;
                    }
                }
                return false;
            }
        }
    }
    return true;
}

3. Tras resolver automáticamente y aceptar, entrar de nuevo al modo competitivo (2) y verificar que:

   * `SOLUCION` esté cargada (se puede mostrar con backdoor 17).
   * Comenzar con la partida ya completa dispara inmediatamente la detección de victoria.

#### 6. Carga de Partida con Estadísticas (v:0.9.3)

1. Iniciar una partida y cometer errores lógicos hasta tener `intentosFallidos > 0`.
2. Guardar la partida (opción 6) y abrir el archivo `.txt` generado.

   * Verificar que incluya:

     * Línea `Nivel:X`
     * 9 líneas de `MATRIZ`
     * 9 líneas de `CELDAFIJA`
     * Línea `// ---- DATOS INTERNOS ----`
     * 9 líneas de `SOLUCION`
     * Línea `INTENTOS: n` con `n > 0`
3. Salir del programa, volver a iniciar y elegir opción 5 (cargar).

   * Al seleccionar el archivo anterior, debe mostrar:

     * `✅ Estadísticas cargadas` si `INTENTOS > 0` o `SODOKU_RESUELTO=true`
     * `⚠️ Sin estadísticas previas` si `INTENTOS=0` y no está resuelto.
4. Repetir la carga con un archivo antiguo (sin bloque interno).

   * Debe inicializar `SOLUCION` en ceros, `intentosFallidos=0` y mostrar `⚠️ Sin estadísticas previas`.
5. Probar carga con archivo corrupto en la sección `SOLUCION` (menos de 9 valores por fila).

   * Debe abortar la carga y mantener el estado previo.

## 📄 CONVENCIONES PARA ARCHIVOS `.h`

* No incluir `using namespace std;` en headers.
* Evitar headers STL (`<vector>`, `<algorithm>`, etc.) salvo que sea imprescindible.
* Siempre usar directivas de inclusión segura:

```cpp
#ifndef NOMBRE_HEADER_H
#define NOMBRE_HEADER_H
// declaraciones
#endif
```

* Incluir únicamente:

  * Declaraciones de funciones
  * Variables `extern`
  * Estructuras o constantes compartidas

---

## 📄 CONVENCIONES PARA ARCHIVOS `.cpp`

* Incluir headers STL **solo si se utilizan directamente**.
* No repetir funciones ya declaradas en los `.h`.
* Las funciones **privadas** deben declararse con forward-declaration si se llaman antes de su definición.
* Agrupar funciones por temática (p.ej. I/O, lógica de juego, utilidades).
* Colocar funciones auxiliares al final del archivo si no son reutilizadas externamente.

---

## 🧪 VALIDACIÓN Y ESTRUCTURA DE ARCHIVOS DE PARTIDA

* Cada archivo `.txt` de partida contiene exactamente **18 líneas**:

  1. 9 líneas de `MATRIZ` (valores de 0 a 9)
  2. 9 líneas de `CELDAFIJA` (valores 0 o 1)

* No usar encabezados como `"MATRIZ:"` o `"CELDAFIJA:"`.

* El nombre de archivo debe usar el formato:

```
YYYYMMDD-HHMMSS.txt
```

* Durante la carga:

  * Validar estructura (filas, columnas, valores válidos).
  * Validar lógica (sin repetidos en filas, columnas o subcuadros 3×3).
  * Rechazar el archivo si falla cualquiera de esas condiciones.

---

## 📩 CONVENCIÓN AMPLIADA – GUARDADO EXTENDIDO Y ESTADÍSTICAS (v:0.9.3)

* **Formato de archivo de partida** actualizado:

  1. Línea de nivel:

     ```
     Nivel:X
     ```

     (0 = modo libre, 1 = fácil, 2 = media, 3 = difícil)
  2. 9 líneas de `MATRIZ` (valores 0 a 9).
  3. 9 líneas de `CELDAFIJA` (valores 0 o 1).
  4. **Bloque adicional opcional**:

     ```
     // ---- DATOS INTERNOS ----
     [SOLUCION 9×9]
     INTENTOS: <n>
     ```

* **Carga (`cargarPartidaDesdeArchivo`)**:

  * Si el bloque adicional existe, se cargan `SOLUCION` e `intentosFallidos`.
  * Mensaje “✅ Estadísticas cargadas” si hay intentos o sudoku resuelto; “⚠️ Sin estadísticas previas” en caso contrario.

* **Guardado (`guardarPartidaEnArchivo`)**:

  * Siempre incluye el bloque de datos internos, aunque esté vacío (intentos = 0).

* **Compatibilidad**:

  * El sistema es retrocompatible con partidas antiguas (sin bloque interno).
  * Si no existe el bloque, `SOLUCION` se inicializa a ceros e `intentosFallidos` se pone en 0.
